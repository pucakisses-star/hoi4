SWE_recalc_parliament_icon_frames = {
	
	set_temp_variable = { forsta_kammaren_temp = forsta_kammaren_support }
	divide_temp_variable = { forsta_kammaren_temp = forsta_kammaren_total }
	multiply_temp_variable = { forsta_kammaren_temp = 11 }
	if = {
		limit = { check_variable = { forsta_kammaren_temp > 0 } }
		clamp_temp_variable = {
			var = forsta_kammaren_temp
			min = 1
			max = 11
		}
	}
	else = {
		set_temp_variable = { forsta_kammaren_temp = 1 }
	}
	set_variable = { SWE_current_forsta_kammaren_icon_frame = forsta_kammaren_temp }

	set_temp_variable = { andra_kammaren_temp = andra_kammaren_support }
	divide_temp_variable = { andra_kammaren_temp = andra_kammaren_total }
	multiply_temp_variable = { andra_kammaren_temp = 25 }
	if = {
		limit = { check_variable = { andra_kammaren_temp > 0 } }
		clamp_temp_variable = {
			var = andra_kammaren_temp
			min = 1
			max = 25
		}
	}
	else = {
		set_temp_variable = { andra_kammaren_temp = 1 }		
	}
	set_variable = { SWE_current_andra_kammaren_icon_frame = andra_kammaren_temp }

	SWE_parliament_support_majority = yes
}


SWE_forsta_kammaren_support = {
	custom_effect_tooltip = SWE_forsta_kammaren_support_tt
	set_temp_variable = { temp_factor = random }
	multiply_temp_variable = { temp_factor = 10 }
	clamp_temp_variable = { var = temp_factor max = 10 min = 1 }
	add_to_variable = { forsta_kammaren_support = temp_factor }
	round_variable = forsta_kammaren_support
	clamp_variable = { var = forsta_kammaren_support max = forsta_kammaren_total min = 0 }
	SWE_recalc_parliament_icon_frames = yes
}

SWE_parliament_support_majority = {
	if = {
		limit = { check_variable = { forsta_kammaren_support > 75 } }
		set_country_flag = forsta_kammaren_support_majority
	}
	else = {
		clr_country_flag = forsta_kammaren_support_majority
	}
	if = {
		limit = { check_variable = { andra_kammaren_support > 116 } }
		set_country_flag = andra_kammaren_support_majority
	}
	else = {
		clr_country_flag = andra_kammaren_support_majority
	}
}


SWE_increase_monarch_power = {
	custom_effect_tooltip = SWE_increase_monarch_power_tt
	hidden_effect = {
		add_to_variable = {
			var = monarch_power_level
			value = 15
		}
		SWE_calculate_monarch_power = yes
	}
}
SWE_decrease_monarch_power = {
	custom_effect_tooltip = SWE_decrease_monarch_power_tt
	hidden_effect = {
		subtract_from_variable = {
			var = monarch_power_level
			value = 15
		}
		SWE_calculate_monarch_power = yes
	}
}
SWE_calculate_monarch_power = {
	clamp_variable = {
		var = monarch_power_level
		min = 0
		max = 100
	}
	set_temp_variable = { monarch_power_location_temp = monarch_power_level }
	multiply_temp_variable = { monarch_power_location_temp = 4 }
	add_to_temp_variable = { monarch_power_location_temp = 35 }
	set_variable = { monarch_power_location = monarch_power_location_temp }
}
