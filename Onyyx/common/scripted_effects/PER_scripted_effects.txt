PER_iranian_decadence_up = {
	custom_effect_tooltip = PER_iranian_decadence_up_tt
	if = { 
		limit = {
			has_idea = PER_iranian_decadence
		}
		swap_ideas = {
			remove_idea = PER_iranian_decadence
			add_idea = PER_iranian_decadence_2
		}	
	}
	else_if = { 
		limit = {
			has_idea = PER_iranian_decadence_2
		}
		swap_ideas = {
			remove_idea = PER_iranian_decadence_2
			add_idea = PER_iranian_decadence_3
		}	
	}
	else_if = { 
		limit = {
			has_idea = PER_iranian_decadence_3
		}
		swap_ideas = {
			remove_idea = PER_iranian_decadence_3
			add_idea = PER_iranian_decadence_4
		}	
	}
	else_if = { 
		limit = {
			has_idea = PER_iranian_decadence_4
		}
		swap_ideas = {
			remove_idea = PER_iranian_decadence_4
			add_idea = PER_iranian_decadence_5
		}	
	}
	else_if = { 
		limit = {
			has_idea = PER_iranian_decadence_5
		}
		remove_ideas = PER_iranian_decadence_5
	}
}
