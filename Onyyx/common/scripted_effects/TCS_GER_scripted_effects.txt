GER_recalc_parliament_icon_frames = {
	
	set_temp_variable = { bundesrat_temp = bundesrat_support }
	divide_temp_variable = { bundesrat_temp = bundesrat_total }
	multiply_temp_variable = { bundesrat_temp = 11 }
	if = {
		limit = { check_variable = { bundesrat_temp > 0 } }
		clamp_temp_variable = {
			var = bundesrat_temp
			min = 1
			max = 11
		}
	}
	else = {
		set_temp_variable = { bundesrat_temp = 1 }
	}
	set_variable = { GER_current_bundesrat_icon_frame = bundesrat_temp }

	set_temp_variable = { reichstag_temp = reichstag_support }
	divide_temp_variable = { reichstag_temp = reichstag_total }
	multiply_temp_variable = { reichstag_temp = 25 }
	if = {
		limit = { check_variable = { reichstag_temp > 0 } }
		clamp_temp_variable = {
			var = reichstag_temp
			min = 1
			max = 25
		}
	}
	else = {
		set_temp_variable = { reichstag_temp = 1 }		
	}
	set_variable = { GER_current_reichstag_icon_frame = reichstag_temp }
}

GER_bundesrat_support = {
	custom_effect_tooltip = tcs_hol_bundesrat_support_tt
	set_temp_variable = { temp_factor = random }
	multiply_temp_variable = { temp_factor = 5 }
	add_to_variable = { bundesrat_support = temp_factor }
	round_variable = bundesrat_support
	clamp_variable = { var = bundesrat_support max = bundesrat_total min = 0 }
	GER_recalc_parliament_icon_frames = yes
}
GER_reichstag_support = {
	custom_effect_tooltip = tcs_hol_reichstag_support_tt
	set_temp_variable = { temp_factor = random }
	multiply_temp_variable = { temp_factor = 10 }
	add_to_variable = { reichstag_support = temp_factor }
	round_variable = reichstag_support
	clamp_variable = { var = reichstag_support max = reichstag_total min = 0 }
	GER_recalc_parliament_icon_frames = yes
}
