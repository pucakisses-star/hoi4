HOL_recalc_parliament_icon_frames = {
	
	set_temp_variable = { eerste_kamer_temp = eerste_kamer_support }
	divide_temp_variable = { eerste_kamer_temp = eerste_kamer_total }
	multiply_temp_variable = { eerste_kamer_temp = 11 }
	if = {
		limit = { check_variable = { eerste_kamer_temp > 0 } }
		clamp_temp_variable = {
			var = eerste_kamer_temp
			min = 1
			max = 11
		}
	}
	else = {
		set_temp_variable = { eerste_kamer_temp = 1 }
	}
	set_variable = { HOL_current_eerste_kamer_icon_frame = eerste_kamer_temp }

	set_temp_variable = { tweede_kamer_temp = tweede_kamer_support }
	divide_temp_variable = { tweede_kamer_temp = tweede_kamer_total }
	multiply_temp_variable = { tweede_kamer_temp = 21 }
	if = {
		limit = { check_variable = { tweede_kamer_temp > 0 } }
		clamp_temp_variable = {
			var = tweede_kamer_temp
			min = 1
			max = 21
		}
	}
	else = {
		set_temp_variable = { tweede_kamer_temp = 1 }		
	}
	set_variable = { HOL_current_tweede_kamer_icon_frame = tweede_kamer_temp }
}

HOL_eerste_kamer_support = {
	custom_effect_tooltip = tcs_hol_eerste_kamer_support_tt
	set_temp_variable = { temp_factor = random }
	multiply_temp_variable = { temp_factor = 5 }
	add_to_variable = { eerste_kamer_support = temp_factor }
	round_variable = eerste_kamer_support
	clamp_variable = { var = eerste_kamer_support max = eerste_kamer_total min = 0 }
	HOL_recalc_parliament_icon_frames = yes
}
HOL_tweede_kamer_support = {
	custom_effect_tooltip = tcs_hol_tweede_kamer_support_tt
	set_temp_variable = { temp_factor = random }
	multiply_temp_variable = { temp_factor = 10 }
	add_to_variable = { tweede_kamer_support = temp_factor }
	round_variable = tweede_kamer_support
	clamp_variable = { var = tweede_kamer_support max = tweede_kamer_total min = 0 }
	HOL_recalc_parliament_icon_frames = yes
}
