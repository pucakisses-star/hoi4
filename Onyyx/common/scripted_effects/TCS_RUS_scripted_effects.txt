
RUS_increase_revolutionary_agitation = {
	if = {
		limit = { has_idea = RUS_revolutionary_agitation_a }
		add_political_power = -50
	}
	else_if = {
		limit = { has_idea = RUS_revolutionary_agitation_b }
		swap_ideas = {
			remove_idea = RUS_revolutionary_agitation_b
			add_idea = RUS_revolutionary_agitation_a
		}
	}
	else_if = {
		limit = { has_idea = RUS_revolutionary_agitation_c }
		swap_ideas = {
			remove_idea = RUS_revolutionary_agitation_c
			add_idea = RUS_revolutionary_agitation_b
		}
	}
	else_if = {
		limit = { has_idea = RUS_revolutionary_agitation_d }
		swap_ideas = {
			remove_idea = RUS_revolutionary_agitation_d
			add_idea = RUS_revolutionary_agitation_c
		}		
	}
	else_if = {
		limit = { has_idea = RUS_revolutionary_agitation_e }
		swap_ideas = {
			remove_idea = RUS_revolutionary_agitation_e
			add_idea = RUS_revolutionary_agitation_d
		}
	}
}
