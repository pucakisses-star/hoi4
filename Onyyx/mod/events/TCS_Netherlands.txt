country_event = { #Country in Chaos Optimized(Fires the Civil War)
    id = netherlands.5
    title = netherlands.5.title
    desc = netherlands.5.desc
    picture = GFX_event_HOL_country_in_chaos
    
    #is_triggered_only = yes
    trigger = {
        has_country_flag = HOL_first_unrest
        has_country_flag = HOL_second_unrest
        OR = {
            has_completed_focus = HOL_united_kuyper
            has_completed_focus = HOL_united_army
        }
    }

    fire_only_once = yes

    option = {
        name = netherlands.5.a
        set_country_flag = HOL_civil_war
        hidden_effect = {
            WRN = {
                transfer_state = 7
            }
            start_civil_war = {
				ideology = rev_socialism
				keep_unit_leaders_trigger = { always = yes }
                size = 0.20
                army_ratio = 0.50
				navy_ratio = 0.75
				states = { 856 7 }
            }
            random_other_country = {
				limit = {
					original_tag = HOL
					has_government = rev_socialism
				}
				WRN = { 
                    annex_country = {
                        target = PREV
                        transfer_troops = yes
                    }
                }
			}
            WRN = {
                declare_war_on = {
                    target = HOL
                    type = civil_war
                }
            }
            WRN = {
                #set_province_controller = 9309
                #set_province_controller = 11318
                set_province_controller = 6286
            }
            HOL = {
                set_province_controller = 3211
                set_province_controller = 391
                set_province_controller = 6336
                set_province_controller = 9335
                set_province_controller = 13772
            }
            every_state = {
				if = {
					limit = { is_core_of = HOL }
					add_core_of = WRN
				}
			}
            WRN = {
                country_event = { 
                    id = netherlands_socialist.1 
                    hours = 2
                }
            }
            HOL = {
                country_event = { 
                    id = netherlands.6
                    hours = 2
                }
            }
        }
    }
    option = {
        name = netherlands.5.b
        set_country_flag = HOL_civil_war
        hidden_effect = {
            WRN = {
                transfer_state = 7
            }
            start_civil_war = {
				ideology = rev_socialism
				keep_unit_leaders_trigger = { always = yes }
				size = 0.50
				navy_ratio = 0.75
				states = { 856 7 }
            }
            random_other_country = {
				limit = {
					original_tag = HOL
					has_government = rev_socialism
				}
				WRN = { 
                    annex_country = {
                        target = PREV
                        transfer_troops = yes
                    }
                }
			}
            WRN = {
                declare_war_on = {
                    target = HOL
                    type = civil_war
                }
            }
            WRN = {
                #set_province_controller = 9309
                #set_province_controller = 11318
                set_province_controller = 6286
                set_province_controller = 68
                set_province_controller = 3314
            }
            HOL = {
                set_province_controller = 3211
                set_province_controller = 391
                set_province_controller = 6336
                set_province_controller = 9335
                set_province_controller = 13772
            }
            every_state = {
				if = {
					limit = { is_core_of = HOL }
					add_core_of = WRN
				}
            }
            WRN = {
                country_event = { 
                    id = netherlands_socialist.1 
                    hours = 1
                }
            }
            WRN = {
                #load_oob = WRN_revolt
                change_tag_from = HOL

                    id = netherlands.6
