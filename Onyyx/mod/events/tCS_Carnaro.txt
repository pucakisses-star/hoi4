country_event = { #Impresa di Fiume
    id = crn.1
    title = crn.1.t
    desc = crn.1.d
    picture = GFX_event_impresa_di_fiume
	
    trigger = {
		NOT = {
			tag = ITA
		}
		NOT = {
			tag = CRN
		}
		NOT = {
			tag = AUS
		}
		owns_state = 848
        date > 1919.8.1
    }

    fire_only_once = yes
	
	immediate = {
        hidden_effect = {
			set_country_flag = dannunzio_happened
			CRN = {
				set_country_flag = impresa_di_fiume
				transfer_state = 848
				load_oob = "CRN_impresa_di_fiume"
				set_division_template_lock = {
					division_template = "Legionari"
					is_locked = yes
				}

			}
		}
	}

    option = {
        name = crn.1.a
        custom_effect_tooltip = crn.1.a.tooltip
    }
}
